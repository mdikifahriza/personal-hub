-- =========================================================
-- RPC: mark task as paid and create linked income transaction
-- =========================================================
DROP FUNCTION IF EXISTS public.mark_task_as_paid(uuid, numeric, date, uuid, uuid, text);

CREATE OR REPLACE FUNCTION public.mark_task_as_paid(
  p_task_id uuid,
  p_paid_amount numeric,
  p_paid_date date DEFAULT CURRENT_DATE,
  p_account_id uuid DEFAULT NULL,
  p_category_id uuid DEFAULT NULL,
  p_description text DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
AS $func$
DECLARE
  v_task_title text;
  v_existing_tx_id uuid;
  v_tx_id uuid;
  v_account_id uuid;
  v_category_id uuid;
BEGIN
  IF p_paid_amount IS NULL OR p_paid_amount <= 0 THEN
    RAISE EXCEPTION 'p_paid_amount must be > 0';
  END IF;

  SELECT t.title
  INTO v_task_title
  FROM public.tasks t
  WHERE t.id = p_task_id;

  IF v_task_title IS NULL THEN
    RAISE EXCEPTION 'Task % not found', p_task_id;
  END IF;

  SELECT tr.id
  INTO v_existing_tx_id
  FROM public.transactions tr
  WHERE tr.source_type = 'task_payment'
    AND tr.source_id = p_task_id
  LIMIT 1;

  IF v_existing_tx_id IS NOT NULL THEN
    RETURN v_existing_tx_id;
  END IF;

  SELECT COALESCE(p_account_id, tf.account_id)
  INTO v_account_id
  FROM public.task_finance tf
  WHERE tf.task_id = p_task_id;

  IF v_account_id IS NULL THEN
    SELECT a.id
    INTO v_account_id
    FROM public.accounts a
    WHERE a.is_active = true
    ORDER BY a.created_at
    LIMIT 1;
  END IF;

  IF v_account_id IS NULL THEN
    RAISE EXCEPTION 'No active account available for task payment';
  END IF;

  SELECT COALESCE(p_category_id, tf.category_id)
  INTO v_category_id
  FROM public.task_finance tf
  WHERE tf.task_id = p_task_id;

  IF v_category_id IS NULL THEN
    SELECT c.id
    INTO v_category_id
    FROM public.categories c
    WHERE c.type = 'income' AND lower(c.name) = 'task payment'
    LIMIT 1;
  END IF;

  IF v_category_id IS NULL THEN
    SELECT c.id
    INTO v_category_id
    FROM public.categories c
    WHERE c.type = 'income'
    ORDER BY c.created_at
    LIMIT 1;
  END IF;

  IF v_category_id IS NULL THEN
    RAISE EXCEPTION 'No income category available';
  END IF;

  INSERT INTO public.transactions (
    type,
    amount,
    category_id,
    account_id,
    description,
    date,
    source_type,
    source_id,
    is_auto_generated
  ) VALUES (
    'income',
    p_paid_amount,
    v_category_id,
    v_account_id,
    COALESCE(p_description, format('Task payment: %s', v_task_title)),
    p_paid_date,
    'task_payment',
    p_task_id,
    true
  )
  RETURNING id INTO v_tx_id;

  INSERT INTO public.task_finance (
    task_id,
    agreed_amount,
    paid_amount,
    paid_at,
    payment_status,
    invoice_status,
    account_id,
    category_id,
    transaction_id,
    updated_at
  ) VALUES (
    p_task_id,
    p_paid_amount,
    p_paid_amount,
    now(),
    'paid',
    'paid',
    v_account_id,
    v_category_id,
    v_tx_id,
    now()
  )
  ON CONFLICT (task_id) DO UPDATE
  SET
    agreed_amount = COALESCE(public.task_finance.agreed_amount, EXCLUDED.agreed_amount),
    paid_amount = EXCLUDED.paid_amount,
    paid_at = EXCLUDED.paid_at,
    payment_status = 'paid',
    invoice_status = 'paid',
    account_id = COALESCE(public.task_finance.account_id, EXCLUDED.account_id),
    category_id = COALESCE(public.task_finance.category_id, EXCLUDED.category_id),
    transaction_id = EXCLUDED.transaction_id,
    updated_at = now();

  UPDATE public.tasks
  SET
    status = 'completed',
    completed_at = COALESCE(completed_at, now())
  WHERE id = p_task_id;

  RETURN v_tx_id;
END;
$func$;

-- =========================================================
-- View: income goal progress
-- =========================================================
CREATE OR REPLACE VIEW public.v_income_goal_progress AS
SELECT
  g.id,
  g.title,
  g.description,
  g.cadence,
  g.target_amount,
  g.start_date,
  g.end_date,
  g.is_active,
  COALESCE(SUM(t.amount), 0)::numeric(15,2) AS achieved_amount,
  GREATEST(g.target_amount - COALESCE(SUM(t.amount), 0), 0)::numeric(15,2) AS remaining_amount,
  CASE
    WHEN g.target_amount <= 0 THEN 0
    ELSE ROUND((COALESCE(SUM(t.amount), 0) / g.target_amount) * 100, 2)
  END AS progress_percent
FROM public.income_goals g
LEFT JOIN public.transactions t
  ON t.type = 'income'
  AND t.date >= g.start_date
  AND t.date <= g.end_date
GROUP BY g.id;
